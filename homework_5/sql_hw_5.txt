-- Создание таблиц

CREATE TABLE dim_date (
    date_sk INT PRIMARY KEY,
    full_date DATE NOT NULL,
    year INT NOT NULL,
    month INT NOT NULL,
    quarter INT NOT NULL,
    day_of_week VARCHAR(10) NOT NULL
);

CREATE TABLE dim_client (
    client_sk INT PRIMARY KEY,
    source_client_id INT NOT NULL,
    first_name VARCHAR(30) NOT NULL,
    last_name VARCHAR(30) NOT NULL,
    email VARCHAR(40)
);

CREATE TABLE dim_trainer (
    trainer_sk INT PRIMARY KEY,
    source_trainer_id INT NOT NULL,
    first_name VARCHAR(30) NOT NULL,
    last_name VARCHAR(30) NOT NULL,
    specialization VARCHAR(50)
);

CREATE TABLE dim_class (
    class_sk INT PRIMARY KEY,
    source_class_id INT NOT NULL,
    class_name VARCHAR(30) NOT NULL,
    description VARCHAR(40)
);

CREATE TABLE dim_membership_type (
    membership_type_sk INT PRIMARY KEY,
    source_type_id INT NOT NULL,
    type_name VARCHAR(50) NOT NULL,
    price DECIMAL(10,2) NOT NULL
);

CREATE TABLE fact_bookings (
    booking_fact_sk INT PRIMARY KEY,
    date_sk INT NOT NULL REFERENCES dim_date(date_sk),
    client_sk INT NOT NULL REFERENCES dim_client(client_sk),
    trainer_sk INT NOT NULL REFERENCES dim_trainer(trainer_sk),
    class_sk INT NOT NULL REFERENCES dim_class(class_sk),
    membership_type_sk INT REFERENCES dim_membership_type(membership_type_sk),
    duration_minutes INT,
    max_capacity INT,
    lead_time_days INT,
    is_confirmed BOOLEAN DEFAULT FALSE,
    is_cancelled BOOLEAN DEFAULT FALSE,
    is_attended BOOLEAN DEFAULT FALSE
);

CREATE INDEX idx_fact_date ON fact_bookings(date_sk);
CREATE INDEX idx_fact_client ON fact_bookings(client_sk);
CREATE INDEX idx_fact_trainer ON fact_bookings(trainer_sk);
CREATE INDEX idx_fact_class ON fact_bookings(class_sk);
CREATE INDEX idx_fact_membership ON fact_bookings(membership_type_sk);
CREATE INDEX idx_fact_attended ON fact_bookings(is_attended);
CREATE INDEX idx_fact_cancelled ON fact_bookings(is_cancelled);
CREATE INDEX idx_fact_confirmed ON fact_bookings(is_confirmed);
CREATE INDEX idx_fact_lead_time ON fact_bookings(lead_time_days);

-- Запрос 1: Самые популярные занятия
-- Бизнес-вопрос: Какие занятия пользуются наибольшим спросом?

SELECT 
    dc.class_name,
    COUNT(*) AS total_bookings,
    COUNT(*) FILTER (WHERE fb.is_attended) AS total_attendances
FROM fact_bookings fb
JOIN dim_class dc ON dc.class_sk = fb.class_sk
GROUP BY dc.class_sk, dc.class_name
ORDER BY COUNT(*) FILTER (WHERE fb.is_attended) DESC;

-- Запрос 2: Рейтинг тренеров по загруженности
-- Бизнес-вопрос: Какой тренер проводит больше всего занятий?

SELECT 
    dt.first_name || ' ' || dt.last_name AS trainer_name,
    COUNT(*) AS total_classes,
    COUNT(DISTINCT fb.client_sk) AS unique_clients
FROM fact_bookings fb
JOIN dim_trainer dt ON dt.trainer_sk = fb.trainer_sk
WHERE fb.is_attended = TRUE
GROUP BY dt.trainer_sk, dt.first_name, dt.last_name
ORDER BY COUNT(*) DESC;

-- Запрос 3: Посещаемость по дням недели
-- Бизнес-вопрос: В какие дни недели клуб посещают чаще всего?

SELECT 
    dd.day_of_week,
    COUNT(*) AS total_bookings,
    COUNT(*) FILTER (WHERE fb.is_attended) AS total_attendances
FROM fact_bookings fb
JOIN dim_date dd ON dd.date_sk = fb.date_sk
GROUP BY dd.day_of_week
ORDER BY COUNT(*) FILTER (WHERE fb.is_attended) DESC;
